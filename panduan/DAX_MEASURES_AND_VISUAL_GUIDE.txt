# KUMPULAN FORMULA DAX UNTUK POWER BI DASHBOARD
# Balai Besar Pelatihan Pertanian (BBPP) Ketindan - Kebun Produksi
# Data Warehouse: Star Schema (Fact_Penjualan, Dim_Waktu, Dim_Produk, Dim_Pelanggan, Dim_Lahan)

================================================================================
1. CORE MEASURES (UKURAN UTAMA)
================================================================================

Total Pendapatan = 
SUM(Fact_Penjualan[TotalPendapatan])

Total Volume Terjual = 
SUM(Fact_Penjualan[Volume])

Total Transaksi = 
COUNTROWS(Fact_Penjualan)

Rata-rata Pendapatan per Transaksi = 
AVERAGE(Fact_Penjualan[TotalPendapatan])

Total Pelanggan Unik = 
DISTINCTCOUNT(Fact_Penjualan[CustomerKey])

Harga Satuan Rata-rata = 
DIVIDE([Total Pendapatan], [Total Volume Terjual], 0)

================================================================================
2. TIME INTELLIGENCE & PERIOD ANALYSIS (KPI 1 & KPI 2)
================================================================================

# Pendapatan Bulan Ini (Agustus / September)
Pendapatan Agustus 2026 = 
CALCULATE(
    [Total Pendapatan],
    Dim_Waktu[NamaBulan] = "Agustus"
)

Pendapatan September 2026 = 
CALCULATE(
    [Total Pendapatan],
    Dim_Waktu[NamaBulan] = "September"
)

# Pertumbuhan Bulanan (MoM Growth %)
Pertumbuhan Pendapatan MoM % = 
VAR RevAgt = [Pendapatan Agustus 2026]
VAR RevSep = [Pendapatan September 2026]
RETURN
DIVIDE(RevSep - RevAgt, RevAgt, 0)

# Rata-rata Pendapatan Mingguan
Rata-rata Pendapatan Mingguan = 
AVERAGEX(
    VALUES(Dim_Waktu[LabelMinggu]),
    [Total Pendapatan]
)

================================================================================
3. RANKING & TOP N METRICS (KPI 3, KPI 4, KPI 5)
================================================================================

# Ranking Produk Berdasarkan Pendapatan (KPI 3)
Rank Produk by Revenue = 
RANKX(
    ALL(Dim_Produk[Komoditas]),
    [Total Pendapatan],
    ,
    DESC,
    Dense
)

# Ranking Produk Berdasarkan Volume Terjual (KPI 4)
Rank Produk by Volume = 
RANKX(
    ALL(Dim_Produk[Komoditas]),
    [Total Volume Terjual],
    ,
    DESC,
    Dense
)

# Ranking Pelanggan Berdasarkan Belanja Terbanyak (KPI 5)
Rank Pembeli by Belanja = 
RANKX(
    ALL(Dim_Pelanggan[NamaPembeli]),
    [Total Pendapatan],
    ,
    DESC,
    Dense
)

# Persentase Kontribusi Produk terhadap Total Pendapatan (Pareto Analysis)
% Kontribusi Pendapatan Produk = 
DIVIDE(
    [Total Pendapatan],
    CALCULATE([Total Pendapatan], ALL(Dim_Produk)),
    0
)

# Persentase Kontribusi Lahan terhadap Total Pendapatan
% Kontribusi Lahan = 
DIVIDE(
    [Total Pendapatan],
    CALCULATE([Total Pendapatan], ALL(Dim_Lahan)),
    0
)

================================================================================
